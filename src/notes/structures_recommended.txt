│   │       └── [id].tsx       <-- Dynamic details (Resolves to: /123)


my-expo-app/
├── src/
│   ├── app/                 <-- All pages and layouts go here
│   │   ├── _layout.tsx      <-- Root layout (e.g., defines Global Providers)
│   │   ├── index.tsx        <-- The App's Home Screen (/)
│   │   ├── (auth)/          <-- Group folder for authentication (hidden from URL)
│   │   │   ├── login.tsx    <-- Resolves to /login
│   │   │   └── register.tsx <-- Resolves to /register
│   │   ├── (tabs)/          <-- Group folder for main navigation tabs
│   │   │   ├── _layout.tsx  <-- Tab configuration file (sets up bottom tabs)
│   │   │   ├── feed.tsx     <-- Resolves to /feed
│   │   │   └── profile.tsx  <-- Resolves to /profile
│   │   └── +not-found.tsx   <-- Fallback for broken links / 404 page
│   │
│   ├── components/          <-- Keep reusable UI out of the app folder!
│   └── hooks/               <-- Reusable React hooks
├── app.json                 <-- Expo configuration settings
└── package.json
